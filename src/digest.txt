import React from 'react';
import { motion } from 'framer-motion';
import type { HTMLMotionProps } from 'framer-motion';

interface ButtonProps extends HTMLMotionProps<"button"> {
    variant?: 'primary' | 'outline' | 'ghost' | 'danger';
    size?: 'sm' | 'md' | 'lg' | 'xl';
    px?: string;
    py?: string;
    className?: string;
}

const Button = React.forwardRef<HTMLButtonElement, ButtonProps>(
    ({ className = '', variant = 'primary', size = 'md', px, py, children, ...props }, ref) => {
        // Shared clip-paths for different sizes
        const polygons = {
            sm: '[clip-path:polygon(2px_0,calc(100%-2px)_0,100%_2px,100%_calc(100%-2px),calc(100%-2px)_100%,2px_100%,0_calc(100%-2px),0_2px)]',
            md: '[clip-path:polygon(4px_0,calc(100%-4px)_0,100%_4px,100%_calc(100%-4px),calc(100%-4px)_100%,4px_100%,0_calc(100%-4px),0_4px)]',
            lg: '[clip-path:polygon(6px_0,calc(100%-6px)_0,100%_6px,100%_calc(100%-6px),calc(100%-6px)_100%,6px_100%,0_calc(100%-6px),0_6px)]',
            xl: '[clip-path:polygon(10px_0,calc(100%-10px)_0,100%_10px,100%_calc(100%-10px),calc(100%-10px)_100%,10px_100%,0_calc(100%-10px),0_10px)]',
        };

        const poly = polygons[size];

        // Button variants handling: Outer background is the border color, Inner is the fill
        const variants = {
            primary: {
                outer: 'bg-white',
                inner: 'bg-white text-black',
            },
            outline: {
                outer: 'bg-white',
                inner: 'bg-black text-white',
            },
            ghost: {
                outer: 'bg-transparent',
                inner: 'bg-transparent text-white group-hover:bg-white/10',
            },
            danger: {
                outer: 'bg-white',
                inner: 'bg-black text-[#FF8000]',
            },
        };

        const currentVariant = variants[variant as keyof typeof variants] || variants.primary;

        const sizes = {
            sm: 'text-xs',
            md: 'text-sm',
            lg: 'text-base',
            xl: 'text-xl md:text-2xl lg:text-3xl',
        };

        const padding = {
            sm: 'p-[1px]', // 1px border
            md: 'p-[2px]', // 2px border
            lg: 'p-[2px]',
            xl: 'p-[2px] md:p-[3px]', // 3px border for giant buttons
        };

        const innerPadding = {
            sm: 'px-3 py-1',
            md: 'px-6 py-2',
            lg: 'px-8 py-3',
            xl: 'px-8 py-3 md:px-10 md:py-4 lg:px-14 lg:py-5',
        };

        const currentInnerPadding = `${px || innerPadding[size].split(' ')[0]} ${py || innerPadding[size].split(' ')[1]}`;

        // Motion variants logic
        const getMotionProps = () => {
            if (variant === 'primary') {
                return {
                    whileHover: {},
                    whileTap: {
                        backgroundColor: '#ef4444',
                        transition: { duration: 0 },
                    },
                };
            }
            if (variant === 'outline') {
                return {
                    whileHover: {
                        opacity: 0.5,
                        transition: { duration: 0.1 },
                    },
                    whileTap: {
                        backgroundColor: '#ef4444',
                        transition: { duration: 0 },
                    },
                };
            }
            if (variant === 'danger') {
                return {
                    whileHover: {
                        opacity: 0.5,
                        transition: { duration: 0.1 },
                    },
                    whileTap: {
                        backgroundColor: '#ef4444',
                        transition: { duration: 0 },
                    },
                };
            }
            return {};
        };

        const getInnerMotionProps = () => {
            if (variant === 'primary') {
                return {
                    whileTap: {
                        backgroundColor: '#ef4444',
                        color: '#ffffff',
                        transition: { duration: 0 },
                    },
                };
            }
            return {};
        };

        return (
            <motion.button
                ref={ref}
                className={`group relative inline-flex items-center justify-center font-bold select-none disabled:!opacity-50 disabled:pointer-events-none uppercase tracking-tighter cursor-pointer ${poly} ${currentVariant.outer} ${padding[size]} ${className}`}
                {...getMotionProps()}
                {...props as HTMLMotionProps<"button">}
            >
                <motion.div
                    className={`w-full h-full flex items-center justify-center ${poly} ${currentVariant.inner} ${sizes[size]} ${currentInnerPadding}`}
                    {...getInnerMotionProps()}
                >
                    {children}
                </motion.div>
            </motion.button>
        );
    }
);

Button.displayName = 'Button';

export default Button;

import { levels } from "../../lib/levels_list";
import Tooltip from "./Tooltip";
import { useMemo } from "react";

interface GraphLevelGuessedProps {
  guessesMap: Map<number, number>;
  correct_level_id: number;
  showLabel?: boolean;
}

const GraphLevelGuessed = ({
  guessesMap,
  correct_level_id,
  showLabel = true,
}: GraphLevelGuessedProps) => {
    const { total, max } = useMemo(() => {
      if (!guessesMap || !(guessesMap instanceof Map)) {
        return { total: 0, max: 0 };
      }
      const values = Array.from(guessesMap.values());
      return {
        total: values.reduce((a, b) => a + b, 0),
          max: values.length > 0 ? Math.max(...values) : 0,
      };
    }, [guessesMap]);

  return (
    <div className="w-full pt-2">
      <div className="flex flex-row items-end h-[40px] w-full gap-[1px]">
        {levels.map((level) => {
          const count = guessesMap.get(level.id) || 0;
          const percent = total > 0 ? Math.round((count / total) * 100) : 0;
          const barHeight = max > 0 ? Math.max(4, (100 * count) / max) : 4;
          const isCorrect = correct_level_id === level.id;

          return (
            <div
              key={level.id}
              className="relative flex flex-col items-center h-full justify-end flex-1"
            >
              <Tooltip
                content={
                  <div className="text-center text-[10px]">
                    <p className="font-bold">{level.levelNumber}</p>
                    <p>{count} Guesses ({percent}%)</p>
                  </div>
                }
                placement="top"
              >
                <div
                  className={`w-full transition-opacity ${
                    isCorrect ? "bg-green-500 opacity-100" : "bg-white/20 opacity-60 hover:opacity-100"
                  }`}
                  style={{ height: `${barHeight}%`, borderRadius: "1px" }}
                />
              </Tooltip>
            </div>
          );
        })}
      </div>
      {showLabel && (
        <p className="text-white/30 text-center text-xs mt-1 uppercase tracking-widest">
          Guess Distribution
        </p>
      )}
    </div>
  );
};

export default GraphLevelGuessed;

import { Link } from "react-router-dom";
import Button from "./Button";
import { useState, useEffect } from "react";

interface Level {
  id: number;
  levelNumber: string;
  name: string;
  thumbnail: string;
}

interface LevelGridProps {
  title: string;
  levels: Level[];
  maxItemsPerRow: number;
  maxImgPerRow?: number;
  layerPath: string;
  counts?: Record<number, number | string>;
  secretLevel?: Level;
  outerPadding?: number;
}

export default function LevelGrid({
  title,
  levels,
  maxItemsPerRow,
  maxImgPerRow,
  layerPath,
  counts = {},
  secretLevel,
  outerPadding = 7,
}: LevelGridProps) {
  const [buttonSize, setButtonSize] = useState<"lg" | "xl">("xl");

  useEffect(() => {
    const handleResize = () => {
      setButtonSize(window.innerWidth < 640 ? "lg" : "xl");
    };
    handleResize();
    window.addEventListener("resize", handleResize);
    return () => window.removeEventListener("resize", handleResize);
  }, []);

  const imgConstraint = maxImgPerRow || maxItemsPerRow;
  const gap = 0.5;
  const pad = outerPadding;
  const sub = (n: number) => (pad + (n - 1) * gap) / n;

  const itemWidth = `max(calc(${100 / maxItemsPerRow}vw - ${
    sub(maxItemsPerRow)
  }rem), 20rem)`;

  const imgWidth = maxImgPerRow
    ? `min(100%, max(calc(${100 / imgConstraint}vw - ${
        sub(imgConstraint) + 4.5
      }rem), 15.5rem))`
    : "100%";

  const sanitizePath = (levelNumber: string, name: string) => {
    return `${levelNumber}-${name}`
      .toLowerCase()
      .replace(/[^a-z0-9\s-]/g, "")
      .replace(/\s+/g, "-");
  };

  return (

      <div className="flex flex-col md:flex-row justify-between items-start md:items-center gap-4 md:gap-0">
        <h1
          className={`text-2xl md:text-6xl uppercase ${

            to={`/${layerPath}/${sanitizePath(
              secretLevel.levelNumber,
              secretLevel.name

          >
            <Button className="mb-2" size={buttonSize} variant="outline">
              <div className="py-1 px-4 md:py-2 md:px-6 text-sm md:text-base">
                SECRET MISSION

            </div>
          </Link>
        )}
      </div>
      <div className="w-full h-0.5 bg-white mb-8" />

      <div className="flex gap-2 flex-wrap justify-center">
        {levels.map((level, idx) => {

import { motion, AnimatePresence } from "framer-motion";
import { useEffect, type ReactNode } from "react";
import { createPortal } from "react-dom";
import Button from "./Button";

interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  title?: string;
  children: ReactNode;
  showCloseButton?: boolean;
  allowBackdropClose?: boolean;
  showFooterButton?: boolean;
  footerButtonText?: string;
  footerText?: string;
  maxWidth?: string;
}

const outerClip =
  "[clip-path:polygon(16px_0,calc(100%-16px)_0,100%_16px,100%_calc(100%-16px),calc(100%-16px)_100%,16px_100%,0_calc(100%-16px),0_16px)]";
const innerClip =
  "[clip-path:polygon(12px_0,calc(100%-12px)_0,100%_12px,100%_calc(100%-12px),calc(100%-12px)_100%,12px_100%,0_calc(100%-12px),0_12px)]";

const Modal = ({
  isOpen,
  onClose,
  title,
  footerText,
  children,
  showCloseButton = true,
  allowBackdropClose = true,

import { useState, type ReactNode, cloneElement, isValidElement } from "react";
import {
    useFloating,
    autoUpdate,
    offset,
    flip,
    shift,
    useHover,
    useFocus,
    useDismiss,
    useRole,
    useInteractions,
    useClick,
    FloatingPortal,
} from "@floating-ui/react";

import type { Placement } from "@floating-ui/react";
import { AnimatePresence, motion } from "framer-motion";

interface TooltipProps {
    content: ReactNode;
    children: ReactNode;
    placement?: Placement;
    className?: string;
    wrapperClassName?: string;
}

const tooltipOuterClip =
    "[clip-path:polygon(6px_0,calc(100%-6px)_0,100%_6px,100%_calc(100%-6px),calc(100%-6px)_100%,6px_100%,0_calc(100%-6px),0_6px)]";
const tooltipInnerClip =
    "[clip-path:polygon(4px_0,calc(100%-4px)_0,100%_4px,100%_calc(100%-4px),calc(100%-4px)_100%,4px_100%,0_calc(100%-4px),0_4px)]";

export default function Tooltip({
    content,
    children,
    placement = "top",
    className = "",
    wrapperClassName = "w-max",
}: TooltipProps) {
    const [isOpen, setIsOpen] = useState(false);

    const { refs, floatingStyles, context } = useFloating({
        open: isOpen,
        onOpenChange: setIsOpen,
        placement,
        whileElementsMounted: autoUpdate,
        middleware: [
            offset(8),
            flip({
                fallbackAxisSideDirection: "start",
            }),
            shift({ padding: 8 }),
        ],
    });

    const hover = useHover(context, { move: false });
    const focus = useFocus(context);
    const dismiss = useDismiss(context);
    const role = useRole(context, { role: "tooltip" });
    const click = useClick(context);

    // Apply interactions
    const { getReferenceProps, getFloatingProps } = useInteractions([
        hover,
        focus,
        dismiss,
        role,
        click, // Adding click gives robust tap behavior on mobile
    ]);

    // Clone element so we don't introduce a wrapper div which can mess with hover hitboxes
    const referenceElement = isValidElement(children) ? (
        cloneElement(children as React.ReactElement<Record<string, unknown>>, {
            ref: refs.setReference,
            ...getReferenceProps({
                ...(children.props as Record<string, unknown>),
            }),
        })
    ) : (
        <span ref={refs.setReference} {...getReferenceProps()} className={wrapperClassName}>
            {children}
        </span>
    );


    return (
        <>
            {referenceElement}
            <FloatingPortal>
                <AnimatePresence>
                    {isOpen && (
                        <div
                            // eslint-disable-next-line react-hooks/refs
                            ref={refs.setFloating}
                            style={floatingStyles}
                            className={`z-[150] pointer-events-none`}
                        >
                            <motion.div
                                {...getFloatingProps()}
                                initial={{ opacity: 0, scale: 0.95, y: placement.startsWith("top") ? 5 : placement.startsWith("bottom") ? -5 : 0 }}
                                animate={{ opacity: 1, scale: 1, y: 0 }}
                                exit={{ opacity: 0, scale: 0.95, y: placement.startsWith("top") ? 5 : placement.startsWith("bottom") ? -5 : 0 }}
                                transition={{ duration: 0.15, ease: "easeOut" }}
                                className={`p-[2px] bg-white pointer-events-auto ${tooltipOuterClip} ${className}`}
                            >
                                <div
                                    className={`bg-black text-white px-3 py-1.5 text-sm font-bold uppercase tracking-tighter relative overflow-hidden ${tooltipInnerClip}`}
                                >
                                    <div className="absolute inset-0 pointer-events-none opacity-[0.05] bg-[linear-gradient(rgba(18,16,16,0)_50%,rgba(0,0,0,0.25)_50%),linear-gradient(90deg,rgba(255,0,0,0.06),rgba(0,255,0,0.02),rgba(0,0,255,0.06))] z-10 bg-[length:100%_2px,3px_100%]" />
                                    <div className="relative z-20 max-w-[60vw]">
                                        {content}
                                    </div>
                                </div>
                            </motion.div>
                        </div>
                    )}
                </AnimatePresence>
            </FloatingPortal>
        </>
    );
}

  storage_path: string;
  times_picked: number;
  avg_score: number;
  avg_time: number;
  created_at: string;
  level_id: number;
  submitter: {
    discord_name: string;
    discord_avatar_url: string;
  };
  guesses?: {
    level_id: number;
    amount: number;
  }[];
}

export default function LevelGallery() {
  const { layerPath, levelName, imgId } = useParams();
  const navigate = useNavigate();
  const [loading, setLoading] = useState(true);
  const [submissions, setSubmissions] = useState<Submission[]>([]);
  const [selectedSubData, setSelectedSubData] = useState<Submission | null>(null);
  const [loadingGuesses, setLoadingGuesses] = useState(false);
  const scrollRefs = useRef<{ [key: string]: HTMLAnchorElement | null }>({});

  const currentSub = submissions.find((s) => s.id.toString() === imgId);
  const ogImage = currentSub 
    ? `https://images.ultrakidle.online/${currentSub.storage_path}` 
    : undefined;

  const level = levels.find((l) => {
    const slug = `${l.levelNumber}-${l.name}`
      .toLowerCase()
      .replace(/[^a-z0-9\s-]/g, "")
      .replace(/\s+/g, "-");
    return slug === levelName;
  });

  useEffect(() => {
    if (!level) return;

    async function fetchLevelData() {
      const { data, error } = await supabase
      .from("image_submissions")
      .select(
        `
        id,
        storage_path,
        times_picked,
        avg_score,
        avg_time,
        created_at,
        level_id,
        submitter:submitter_profiles (
          discord_name,
          discord_avatar_url
        )
        `
      )
      .eq("level_id", level!.id)
      .eq("status", "approved")
      .order("created_at", { ascending: false });

      if (!error && data) setSubmissions(data as unknown as Submission[]);
      setLoading(false);
    }

    fetchLevelData();
  }, [level]);

  useEffect(() => {
    if (!imgId || submissions.length === 0) {
      // eslint-disable-next-line react-hooks/set-state-in-effect
      setSelectedSubData(null);
      return;
    }

    const baseSub = submissions.find((s) => s.id.toString() === imgId);
    if (!baseSub) {
      navigate(`/${layerPath}/${levelName}`);
      return;
    }

    async function fetchGuesses(sub: Submission) {
      setLoadingGuesses(true);
      const { data, error } = await supabase
        .from("inferno_image_guesses")
        .select("level_id:id_level, amount")
        .eq("id_image_submit", imgId);

      if (!error && data) {
        setSelectedSubData({
          ...sub,
          guesses: data as unknown as Submission["guesses"],
        });
      } else {
        setSelectedSubData({ ...sub, guesses: [] });
      }
      setLoadingGuesses(false);
    }

    fetchGuesses(baseSub);
  }, [imgId, submissions, navigate, layerPath, levelName]);

  useEffect(() => {
    if (!loading && imgId && scrollRefs.current[imgId]) {
      scrollRefs.current[imgId]?.scrollIntoView({
        behavior: "smooth",
        block: "center",
      });
    }
  }, [loading, imgId]);

  if (!level) {
    return (
      <PageLayout title="NOT FOUND" backPath={`/${layerPath}`}>
        <div className="text-center py-20 text-2xl uppercase opacity-50">
          Level not found
        </div>
      </PageLayout>
    );
  }

  const maxItemsPerRow = 3;
  const gap = 0.5;
  const pad = 7;
  const subCalc = (n: number) => (pad + (n - 1) * gap) / n;
  const itemWidth = `max(calc(${100 / maxItemsPerRow}vw - ${subCalc(
    maxItemsPerRow
  )}rem), 20rem)`;

  return (
    <div>
      <SEO
        title={`${level.name.toUpperCase()} - Gallery`}
        description={`View community submissions for ${level.name}`}
        image={ogImage}
      />
      <PageLayout title={level.name.toUpperCase()} backPath={`/${layerPath}`}>
        <div className="flex flex-col gap-2 w-full">
          <div className="flex flex-col md:flex-row gap-6 items-center ">
            <img
              src={level.thumbnail}
              alt={level.name}
              className="w-full md:w-80 aspect-video object-cover "
            />
            <div className="flex flex-col gap-2">
              <h2 className="text-4xl font-bold">{level.levelNumber}</h2>
              <p className="text-xl text-white/70 italic uppercase tracking-tighter">
                {submissions.length} Approved Submissions
              </p>
            </div>
          </div>
          <div className="w-full h-0.5 bg-white mb-3"/>

          {loading ? (
            <div className="text-center py-20 text-2xl animate-pulse uppercase opacity-50">
              Fetching Data...
            </div>
          ) : (
            <div className="flex gap-2 flex-wrap justify-center">
              {submissions.map((sub) => (
                <Link
                  key={sub.id}
                  to={`/${layerPath}/${levelName}/${sub.id}`}
                  ref={(el) => { scrollRefs.current[sub.id] = el; }}
                  style={{ width: itemWidth }}
                  className="relative flex flex-col bg-black/40 p-4 [clip-path:polygon(8px_0,calc(100%-8px)_0,100%_8px,100%_calc(100%-8px),calc(100%-8px)_100%,8px_100%,0_calc(100%-8px),0_8px)]"
                >
                  <div className="flex justify-between items-center mb-2">
                    <div className="flex items-center gap-2">
                      <SubmitterProfilePicture url={sub.submitter.discord_avatar_url} />
                      <span className="text-sm font-bold truncate max-w-[120px]">
                        {sub.submitter.discord_name}

                  <div className="w-full px-5 mt-auto mb-auto flex justify-center">
                    <div className="w-full aspect-video bg-zinc-900 overflow-hidden">
                      <img

import { levels } from "@/lib/levels_list";
import SEO from "@/components/SEO";
import SubmitterProfilePicture from "@/components/ui/SubmitterProfilePicture";

interface Submission {
  id: number;
  storage_path: string;
  times_picked: number;
  avg_score: number;
  avg_time: number;
  created_at: string;
  level_id: number;
  submitter: {
    discord_name: string;
    discord_avatar_url: string;
  };
  guesses?: {
    level_id: number;
    amount: number;
  }[];
}

export default function SubmitterGallery() {
  const { submitterId, imgId } = useParams();
  const navigate = useNavigate();
  const [loading, setLoading] = useState(true);
  const [submissions, setSubmissions] = useState<Submission[]>([]);
  const [selectedSubData, setSelectedSubData] = useState<Submission | null>(null);
  const [loadingGuesses, setLoadingGuesses] = useState(false);
  const scrollRefs = useRef<{ [key: string]: HTMLDivElement | null }>({});

  const currentSub = submissions.find((s) => s.id.toString() === imgId);
  const ogImage = currentSub 
    ? `https://images.ultrakidle.online/${currentSub.storage_path}` 
    : undefined;

  const submitterProfile = submissions[0]?.submitter;

  useEffect(() => {
    async function fetchSubmitterSubmissions() {
      const { data, error } = await supabase
        .from("image_submissions")
        .select(`
          id,
          storage_path,
          times_picked,
          avg_score,
          avg_time,
          created_at,
          level_id,
          submitter:submitter_profiles (
            discord_name,
            discord_avatar_url
          )
        `)
        .eq("submitter_id", submitterId)
        .eq("status", "approved")
        .order("created_at", { ascending: false });

      if (!error && data) {
        setSubmissions(data as unknown as Submission[]);
      }
      setLoading(false);
    }

    fetchSubmitterSubmissions();
  }, [submitterId]);

  useEffect(() => {
    if (!imgId || submissions.length === 0) {
      setSelectedSubData(null);
      return;
    }

    const baseSub = submissions.find((s) => s.id.toString() === imgId);
    if (!baseSub) {
      navigate(`/submitter/${submitterId}`);
      return;
    }

    async function fetchGuesses(sub: Submission) {
      setLoadingGuesses(true);
      const { data, error } = await supabase
        .from("inferno_image_guesses")
        .select("level_id:id_level, amount")
        .eq("id_image_submit", imgId);

      if (!error && data) {
        setSelectedSubData({ ...sub, guesses: data as unknown as Submission["guesses"] });
      } else {
        setSelectedSubData({ ...sub, guesses: [] });
      }
      setLoadingGuesses(false);
    }

    fetchGuesses(baseSub);
  }, [imgId, submissions, navigate, submitterId]);

  useEffect(() => {
    if (!loading && imgId && scrollRefs.current[imgId]) {
      scrollRefs.current[imgId]?.scrollIntoView({
        behavior: "smooth",
        block: "center",
      });
    }
  }, [loading, imgId]);

  const maxItemsPerRow = 3;
  const gap = 0.5;
  const pad = 7;
  const subCalc = (n: number) => (pad + (n - 1) * gap) / n;
  const itemWidth = `max(calc(${100 / maxItemsPerRow}vw - ${subCalc(
    maxItemsPerRow
  )}rem), 20rem)`;

  return (
    <div>
      <div>
        <SEO
          title={`${submitterProfile?.discord_name || "Submitter"} - Gallery`}
          description={`View approved submissions uploaded by ${submitterProfile?.discord_name}`}
          image={ogImage}
        />
      </div>
      <PageLayout title={submitterProfile?.discord_name.toUpperCase() || "SUBMITTER GALLERY"} backPath="/submitter-search">
        <div className="flex flex-col gap-2 w-full">
          {submitterProfile && (
            <div className="flex gap-4 md:gap-6 items-center">
              <SubmitterProfilePicture
                url={submitterProfile.discord_avatar_url}
                className="w-12 h-12 md:w-20 md:h-20 rounded-full border-2 border-white shrink-0"
              />
              <div className="flex flex-col gap-1 min-w-0">
                <h2 className="text-2xl md:text-4xl font-bold uppercase truncate">
                  {submitterProfile.discord_name}
                </h2>
                <p className="text-sm md:text-xl text-white/70 italic uppercase tracking-tighter">
                  {submissions.length} Approved Submissions
                </p>
              </div>
            </div>
          )}
          <div className="w-full h-0.5 bg-white mb-3" />

          {loading ? (
            <div className="text-center py-20 text-2xl animate-pulse uppercase opacity-50">
              Fetching Submissions...
            </div>
          ) : submissions.length === 0 ? (
            <div className="text-center py-20 text-2xl uppercase opacity-50">
              Submitter Not Found
            </div>
          ) : (
            <div className="flex gap-2 flex-wrap justify-center">
              {submissions.map((sub) => {
                const matchedLevel = levels.find((l) => l.id === sub.level_id);
                return (
                  <div
                    key={sub.id}
                    ref={(el) => { scrollRefs.current[sub.id] = el; }}
                    onClick={() => navigate(`/submitter/${submitterId}/${sub.id}`)}
                    style={{ width: itemWidth }}
                    className="relative flex flex-col bg-black/40 p-4 [clip-path:polygon(8px_0,calc(100%-8px)_0,100%_8px,100%_calc(100%-8px),calc(100%-8px)_100%,8px_100%,0_calc(100%-8px),0_8px)] cursor-pointer hover:bg-white/[0.02]"
                  >
                    <div className="flex justify-between items-center mb-2">
                      <span className="text-sm font-bold uppercase truncate max-w-[180px]">
                        {matchedLevel ? `${matchedLevel.levelNumber} ${matchedLevel.name}` : "Unknown Level"}
                      </span>
                      <span className="text-xs text-white/40 uppercase">
                        #{sub.id}
                      </span>
                    </div>

                    <div className="w-full h-0.5 bg-white mb-3" />

                    <div className="w-full px-5 mt-auto mb-auto flex justify-center">
                      <div className="w-full aspect-video bg-zinc-900 overflow-hidden">
                        <img
                          src={`https://images.ultrakidle.online/${sub.storage_path}`}
                          className="w-full h-full object-contain"
                          loading="lazy"
                        />
                      </div>
                    </div>

                    <div className="w-full h-0.5 bg-white mt-3" />

                    <div className="grid grid-cols-3 gap-1 mt-2">
                      <div className="text-center">
                        <p className="text-xs text-white uppercase tracking-tighter">
                          Times Picked
                        </p>
                        <p className="text-sm ">{sub.times_picked}</p>
                      </div>
                      <div className="text-center border-x-2 border-white">
                        <p className="text-xs text-white uppercase tracking-tighter">
                          Avg Score
                        </p>
                        <p className="text-sm ">{Math.round(sub.avg_score)}</p>
                      </div>
                      <div className="text-center">
                        <p className="text-xs text-white uppercase tracking-tighter">
                          Avg Time
                        </p>
                        <p className="text-sm ">
                          {Number(sub.avg_time).toFixed(1)}s
                        </p>
                      </div>
                    </div>
                  </div>
                );
              })}
            </div>
          )}
        </div>

      </PageLayout>
      <Modal
        isOpen={!!imgId}
        onClose={() => navigate(`/submitter/${submitterId}`)}
        maxWidth="max-w-5xl"
        showFooterButton={false}
      >
        <div className="flex flex-col md:flex-row min-h-0 h-full w-full">
          <div className="flex-[2] bg-black flex items-center justify-center md:border-b-0 md:border-r-2 border-white">
            {submissions.find((s) => s.id.toString() === imgId) && (
              <img
                src={`https://images.ultrakidle.online/${
                  submissions.find((s) => s.id.toString() === imgId)
                    ?.storage_path
                }`}
                className="max-w-full max-h-[70vh] object-contain bg-black"
              />
            )}
          </div>

          <div className="flex-1 p-6 flex flex-col gap-6 overflow-y-auto">
            {selectedSubData ? (
              <>
                <div className="flex justify-between items-start">
                  <div className="flex items-center gap-3">
                    <SubmitterProfilePicture
                      url={selectedSubData.submitter.discord_avatar_url}
                      className="w-10 h-10 rounded-full border-2 border-white"
                    />
                    <div>
                      <p className="font-bold text-lg text-white uppercase">
                        {selectedSubData.submitter.discord_name}
                      </p>
                      <p className="text-xs text-white uppercase">
                        {new Date(
                          selectedSubData.created_at
                        ).toLocaleDateString()}
                      </p>
                    </div>
                  </div>
                </div>

                <div className="grid grid-cols-2 md:gap-4 gap-2 border-y-2 border-white md:py-4 py-2">
                  <div className="text-center">
                    <p className="text-sm text-white uppercase md:mb-1">Times Picked</p>
                    <p className="text-lg md:text-2xl ">{selectedSubData.times_picked}</p>
                  </div>
                  <div className="text-center">
                    <p className="text-sm text-white uppercase md:mb-1">Total Guesses</p>
                    <p className="text-lg md:text-2xl ">
                      {selectedSubData.guesses?.reduce((acc, g) => acc + Number(g.amount), 0) ||
                        0}
                    </p>
                  </div>
                  <div className="text-center">
                    <p className="text-sm text-white uppercase md:mb-1">Avg Score</p>
                    <p className="text-lg md:text-2xl ">{Math.round(selectedSubData.avg_score)}</p>
                  </div>
                  <div className="text-center">
                    <p className="text-sm text-white uppercase md:mb-1">Avg Time</p>
                    <p className="text-lg md:text-2xl ">
                      {Number(selectedSubData.avg_time).toFixed(2)}s
                    </p>
                  </div>
                </div>

                <div className="flex-1">
                  {loadingGuesses ? (
                    <div className="animate-pulse text-xs uppercase opacity-50">
                      Loading Stats...
                    </div>
                  ) : (
                    <GraphLevelGuessed
                      guessesMap={
                        new Map(
                          selectedSubData.guesses?.map((g) => [
                            Number(g.level_id),
                            Number(g.amount),
                          ])
                        )
                      }
                      correct_level_id={selectedSubData.level_id}
                    />
                  )}
                </div>
              </>
            ) : (
              <div className="flex items-center justify-center h-full animate-pulse uppercase">
                Loading Stats...
              </div>
            )}
          </div>
        </div>
      </Modal>
    </div>
  );
}

import { useState, useEffect } from "react";
import { Link } from "react-router-dom";
import PageLayout from "@/components/layout/PageLayout";
import SubmitterProfilePicture from "@/components/ui/SubmitterProfilePicture";
import { supabase } from "@/lib/supabase";

interface Submitter {
  id: string;
  discord_name: string;
  discord_avatar_url: string;
  submissions_count: number;
}

export default function SubmitterSearch() {
  const [search, setSearch] = useState("");
  const [submitters, setSubmitters] = useState<Submitter[]>([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    const delayDebounce = setTimeout(() => {
      async function searchSubmitters() {
        setLoading(true);

        const { data, error } = await supabase
          .from("approved_submitters_v")
          .select("id, discord_name, discord_avatar_url, submissions_count")
          .ilike("discord_name", `%${search}%`)
          .order("submissions_count", { ascending: false });

        if (!error && data) {
          const formatted: Submitter[] = data.map((item: any) => ({
            id: item.id,
            discord_name: item.discord_name,
            discord_avatar_url: item.discord_avatar_url,
            submissions_count: item.submissions_count,
          }));
          setSubmitters(formatted);
        }
        setLoading(false);
      }

      searchSubmitters();
    }, 300);

    return () => clearTimeout(delayDebounce);
  }, [search]);

  return (
    <PageLayout title="SUBMITTER SEARCH" backPath="/">
      <div className="flex flex-1 flex-col gap-6 w-full max-w-4xl mx-auto md:h-[calc(100dvh-175px)] h-[calc(100dvh-110px)]">
        <input
          type="text"
          placeholder="SEARCH BY DISCORD NAME..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="border-2 border-white w-full bg-black/40 text-white py-2 px-4 uppercase tracking-tighter text-lg md:text-xl font-vcr outline-none placeholder:text-white/30 shrink-0"
        />

        {loading ? (
          <div className="text-center py-10 text-2xl animate-pulse uppercase opacity-50">
            Searching Submitters...
          </div>
        ) : (
          <div className="grid grid-cols-2 md:grid-cols-4 gap-3 overflow-y-auto pr-1">
            {submitters.map((sub) => (
              <Link
                key={sub.id}
                to={`/submitter/${sub.id}`}
                className="flex items-center gap-3 bg-black/40 hover:bg-white/10 transition-colors p-3 [clip-path:polygon(8px_0,calc(100%-8px)_0,100%_8px,100%_calc(100%-8px),calc(100%-8px)_100%,8px_100%,0_calc(100%-8px),0_8px)] border-b border-transparent "
              >
                <SubmitterProfilePicture
                  url={sub.discord_avatar_url}
                  className="w-10 h-10 rounded-full border border-white/20 shrink-0"
                />
                <div className="flex flex-col min-w-0">
                  <span className="font-bold text-base truncate text-white uppercase">
                    {sub.discord_name}
                  </span>
                  <span className="text-xs text-white/50 uppercase truncate">
                    {sub.submissions_count} Approved
                  </span>
                </div>
              </Link>
            ))}
            {!loading && submitters.length === 0 && (
              <div className="col-span-full text-center py-10 text-xl uppercase opacity-50">
                No submitters found
              </div>
            )}
          </div>
        )}
      </div>
    </PageLayout>
  );
}
